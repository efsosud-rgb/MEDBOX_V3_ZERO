  234: def normalizar_texto_mapa(valor):
  235:     import unicodedata
  236:     texto = str(valor or "").strip().upper()
  237:     texto = unicodedata.normalize("NFD", texto)
  238:     texto = "".join(c for c in texto if unicodedata.category(c) != "Mn")
  239:     texto = texto.replace("Ã‡", "C")
  240:     texto = re.sub(r"\s+", " ", texto)
  241:     return texto.strip()

  244: def converter_float_mapa(valor):
  245:     if valor is None:
  246:         return 0.0
  247: 
  248:     if isinstance(valor, (int, float)):
  249:         return float(valor)
  250: 
  251:     texto = str(valor).strip()
  252:     texto = texto.replace("R$", "").replace(" ", "")
  253: 
  254:     if not texto:
  255:         return 0.0
  256: 
  257:     # Ex.: 1.234,56
  258:     if "," in texto and "." in texto:
  259:         texto = texto.replace(".", "").replace(",", ".")
  260:     else:
  261:         texto = texto.replace(",", ".")
  262: 
  263:     texto = re.sub(r"[^0-9.\-]", "", texto)
  264: 
  265:     try:
  266:         return float(texto)
  267:     except Exception:
  268:         return 0.0
