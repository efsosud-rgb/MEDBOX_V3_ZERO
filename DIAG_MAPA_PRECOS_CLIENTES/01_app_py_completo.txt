   59:     logger = logging.getLogger("importacao")
   60:     logger.setLevel(logging.DEBUG)
   61: 
   62:     for handler in logger.handlers[:]:
   63:         handler.close()
   64:         logger.removeHandler(handler)
   65: 
   66:     file_handler = logging.FileHandler(LOG_PATH, mode="w", encoding="utf-8")
   67:     file_handler.setLevel(logging.DEBUG)
   68: 
   69:     formatter = logging.Formatter(
   70:         fmt="%(asctime)s.%(msecs)03d | %(message)s",
   71:         datefmt="%H:%M:%S",
   72:     )
   73:     file_handler.setFormatter(formatter)
   74: 
   75:     class FlushFileHandler(logging.FileHandler):
   76:         def emit(self, record):
   77:             super().emit(record)
   78:             self.flush()
   79: 
   80:     logger.handlers = []
   81:     fh = FlushFileHandler(LOG_PATH, mode="w", encoding="utf-8")
   82:     fh.setLevel(logging.DEBUG)
   83:     fh.setFormatter(formatter)
   84:     logger.addHandler(fh)
   85:     logger.propagate = False
   86:
